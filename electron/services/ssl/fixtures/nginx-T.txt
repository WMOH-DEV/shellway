    worker_connections 768;
}

    sendfile on;
    # server { listen 443 ssl; server_name commented.example.com; }
    include /etc/nginx/mime.types;

    include /etc/nginx/sites-enabled/*;
}

# configuration file /etc/nginx/mime.types:
types {
    text/html html htm;

# configuration file /etc/nginx/conf.d/status.conf:
server {
    listen 127.0.0.1:8080;
    server_name localhost;

}

# configuration file /etc/nginx/sites-enabled/app.example.com:
# Redirect and TLS for app.example.com
server {
    listen 80;
    listen [::]:80;
    server_name app.example.com www.example.com;
    return 301 https://$host$request_uri;
}

server {
    listen 443 ssl http2; # managed by Certbot
    listen [::]:443 ssl http2;
    server_name app.example.com
                www.example.com;
    ssl_certificate /etc/letsencrypt/live/app.example.com/fullchain.pem; # managed by Certbot
    ssl_certificate_key /etc/letsencrypt/live/app.example.com/privkey.pem; # managed by Certbot

    location / {
        if ($request_uri ~ "\{x\}") { return 404; }
        proxy_pass http://127.0.0.1:8080;
    }
    location ~ ^/(a|b)/ {
        root /var/www/html;
    }
}
